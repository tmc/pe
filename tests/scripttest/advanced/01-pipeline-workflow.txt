# Create a complex pipeline for prompt development
$ echo "Explain quantum computing" > base.txt

# Pipeline: optimize → test → branch → deploy
$ pe optimize base.txt --method pe2 --iterations 3 | \
  pe test --inline --assert "contains:qubit" | \
  pe branch create feature/quantum --auto-commit | \
  pe build --output quantum.prod
