echo '#!/usr/bin/env pe run --provider=openai --model=gpt-4 --temperature=0.7

# Comprehensive Prompt
# Tests all prompt sections and features

Main prompt: Process {{.data}} in {{.format}} format.

---defaults---
data: sample input
format: JSON

---system---
You are a data processing expert.
Always validate input before processing.

---tests---
- name: basic_test
  vars:
    data: "test data"
    format: "XML"
  assert:
    - type: contains
      value: "processed"

---variants---
verbose:
  Provide detailed processing of {{.data}} in {{.format}} format.
  Include step-by-step analysis.

concise:
  Process {{.data}} ({{.format}}): brief output only.

---metadata---
version: 1.0
author: test' > comprehensive.prompt
chmod +x comprehensive.prompt

stdout 'Test.*: 1'
stdout 'Variants: verbose, concise'

# Test prompt help shows system prompt
pe prompt help comprehensive.prompt
stdout 'System Prompt:'
stdout 'data processing expert'
