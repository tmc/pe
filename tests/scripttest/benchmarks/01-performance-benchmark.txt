# Test: Performance benchmarking

# Create benchmark configuration
$ cat > benchmark.yaml << EOF
name: "Sentiment Analysis Benchmark"
iterations: 10
prompts:
  - id: baseline
    text: "Analyze sentiment"
  - id: optimized
    text: "As a sentiment expert, analyze the emotional tone"
  - id: cot
    text: "Think step by step to analyze sentiment"
providers:
  - name: gpt-4
  - name: claude-3
  - name: ollama:llama2
metrics:
  - latency
  - tokens
  - accuracy
EOF
